  const [isVisible, setIsVisible] = useState(false);
  const [activeTab, setActiveTab] = useState("skills");
  const sectionRef = useRef(null);
  const headerRef = useRef(null);
  const profileRef = useRef(null);
  const tabsRef = useRef(null);

  // Skills data
  const skills = [
    { name: "React", level: 90, icon: "⚛️" },
    { name: "JavaScript", level: 85, icon: "🟨" },
    { name: "HTML/CSS", level: 95, icon: "🎨" },
    { name: "Tailwind CSS", level: 88, icon: "🌊" },
    { name: "UI/UX Design", level: 80, icon: "📱" },
    { name: "Node.js", level: 75, icon: "🟢" },
  ];

  // Experience data
  const experiences = [
    {
      period: "2021 - Present",
      role: "Senior Frontend Developer",
      company: "TechVision Inc.",
      description:
        "Lead the frontend development team in creating responsive and accessible web applications using React and modern JavaScript.",
    },
    {
      period: "2019 - 2021",
      role: "UI/UX Designer",
      company: "Creative Solutions",
      description:
        "Designed user interfaces for web and mobile applications, focusing on user experience and visual appeal.",
    },
    {
      period: "2017 - 2019",
      role: "Web Developer",
      company: "Digital Innovations",
      description:
        "Developed and maintained client websites using HTML, CSS, JavaScript, and various content management systems.",
    },
  ];

  // Education data
  const education = [
    {
      period: "2016 - 2017",
      degree: "Master of Computer Science",
      institution: "Tech University",
      description: "Specialized in web technologies and user interface design.",
    },
    {
      period: "2012 - 2016",
      degree: "Bachelor of Science in Information Technology",
      institution: "Digital Arts College",
      description:
        "Graduated with honors. Focus on programming and design fundamentals.",
    },
  ];

  // GSAP Animations
  useEffect(() => {
    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setIsVisible(true);
          observer.unobserve(entry.target);

          // Header animation
          gsap.fromTo(
            headerRef.current,
            { y: 50, opacity: 0 },
            { y: 0, opacity: 1, duration: 1, ease: "power3.out" }
          );

          // Profile animation
          gsap.fromTo(
            profileRef.current,
            { x: -100, opacity: 0 },
            {
              x: 0,
              opacity: 1,
              duration: 1.2,
              ease: "back.out(1.4)",

              opacity: 1,

              delay: 0.5,

          );

    hidden: { opacity: 0, y: 20 },
    visible: {
      opacity: 1,
      y: 0,
      transition: { duration: 0.5, ease: "easeOut" },
    },
    exit: { opacity: 0, y: -20, transition: { duration: 0.3, ease: "easeIn" } },

              <span className="absolute -bottom-2 left-0 w-full h-1 bg-gradient-to-r from-purple-500 to-cyan-500 rounded-full opacity-75 blur-sm"></span>
            </span>
          </h2>
          <p className="text-gray-300 text-lg max-w-3xl mx-auto leading-relaxed">

            developer who blends technical expertise with a passion for design.
          </p>
        </div>

        <div className="flex flex-col lg:flex-row gap-12 items-start">
          {/* Profile Information */}
          <div ref={profileRef} className="w-full lg:w-2/5">
            <div className="bg-gray-800/50 backdrop-blur-md rounded-2xl p-8 border border-gray-700/50 shadow-2xl shadow-purple-900/10 relative">
              <div className="absolute inset-0 bg-gradient-to-br from-purple-500/10 to-cyan-500/10 rounded-2xl -z-10"></div>
              <h3 className="text-2xl font-semibold text-white mb-6 bg-gradient-to-r from-purple-400 to-cyan-400 text-transparent bg-clip-text">
                Who I Am
              </h3>
              <p className="text-gray-200 mb-6 leading-relaxed">
                Hey there! I’m John, a San Francisco-based developer with 5+
                years of experience in building sleek, user-focused web
                applications.
              </p>
              <div className="grid grid-cols-2 gap-6 mb-8">
                <div>
                  <h4 className="text-gray-400 text-sm uppercase tracking-wide">
                    Name
                  </h4>
                  <p className="text-white font-medium">John Doe</p>
                </div>
                <div>
                  <h4 className="text-gray-400 text-sm uppercase tracking-wide">
                    Email
                  </h4>
                  <p className="text-white font-medium">john@example.com</p>
                </div>
                <div>
                  <h4 className="text-gray-400 text-sm uppercase tracking-wide">
                    Location
                  </h4>
                  <p className="text-white font-medium">San Francisco, CA</p>
                </div>
                <div>
                  <h4 className="text-gray-400 text-sm uppercase tracking-wide">
                    Status
                  </h4>
                  <p className="text-white font-medium">Open to Work</p>
                </div>
              </div>
              <button className="w-full py-3 px-6 bg-gradient-to-r from-purple-600 to-cyan-600 text-white font-medium rounded-lg hover:bg-gradient-to-r hover:from-purple-700 hover:to-cyan-700 transition-all duration-300 transform hover:-translate-y-1 shadow-lg hover:shadow-purple-500/30">

              </button>

          </div>

              {["skills", "experience", "education"].map((tab) => (
                <button
                  key={tab}
                  className={`flex-1 py-3 px-6 font-medium rounded-lg transition-all duration-300 relative group ${
                    activeTab === tab
                      ? "text-white bg-gradient-to-r from-purple-600/50 to-cyan-600/50"
                      : "text-gray-400 hover:text-gray-200 hover:bg-gray-700/30"
                  }`}
                  onClick={() => setActiveTab(tab)}
                >
                  <span className="relative z-10 capitalize">{tab}</span>
                  {activeTab === tab && (
                    <span className="absolute inset-0 bg-gradient-to-r from-purple-500 to-cyan-500 opacity-20 rounded-lg blur-sm"></span>
                  )}
                </button>
              ))}
            </div>

            {/* Tab Content */}
            <div className="bg-gray-800/50 backdrop-blur-md rounded-2xl p-8 border border-gray-700/50 shadow-2xl shadow-purple-900/10">
              <AnimatePresence mode="wait">
                {/* Skills Tab */}
                {activeTab === "skills" && (
                  <motion.div
                    key="skills"
                    variants={tabVariants}
                    initial="hidden"
                    animate="visible"
                    exit="exit"
                    className="grid gap-6"
                  >
                    {skills.map((skill, index) => (
                      <div key={index} className="group">
                        <div className="flex items-center justify-between mb-2">
                          <div className="flex items-center gap-3">
                            <span className="text-2xl">{skill.icon}</span>
                            <span className="text-white font-medium">
                              {skill.name}
                            </span>
                          </div>
                          <span className="text-gray-300 text-sm">
                            {skill.level}%
                          </span>
                        </div>
                        <div className="w-full h-2 bg-gray-700/50 rounded-full overflow-hidden">
                          <motion.div
                            className="h-full bg-gradient-to-r from-purple-500 to-cyan-500 rounded-full"
                            initial={{ width: 0 }}
                            animate={{
                              width: isVisible ? `${skill.level}%` : 0,
                            }}
                            transition={{
                              duration: 1.5,
                              ease: "easeOut",
                              delay: index * 0.1,
                            }}
                          />
                        </div>
                      </div>
                    ))}
                  </motion.div>
                )}

                {/* Experience Tab */}
                {activeTab === "experience" && (
                  <motion.div
                    key="experience"
                    variants={tabVariants}
                    initial="hidden"
                    animate="visible"
                    exit="exit"

                        <span className="absolute -left-3 w-6 h-6 bg-cyan-500 rounded-full flex items-center justify-center text-white text-sm">
                          {index + 1}
                        </span>
                        <div className="bg-gray-700/30 rounded-xl p-6 hover:bg-gray-700/50 transition-all duration-300">
                          <span className="inline-block px-3 py-1 bg-cyan-500/20 text-cyan-300 text-xs font-medium rounded-full mb-3">
                            {edu.period}
                          </span>
                          <h4 className="text-xl font-medium text-white mb-1">
                            {edu.degree}
                          </h4>
                          <p className="text-purple-400 mb-3">
                            {edu.institution}
                          </p>
                          <p className="text-gray-200">{edu.description}</p>
                        </div>
                      </div>
                    ))}
                  </motion.div>
                )}
              </AnimatePresence>
            </div>
          </div>
        </div>
      </div>
    </section>
  );
};

export default About;
